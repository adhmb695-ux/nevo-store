const express = require('express'); const path = require('path'); require('dotenv').config(); const app = express(); const PORT = process.env.PORT || 3000; // Middleware لتمرير البيانات المشفّرة وبناء مجلد الملفات الثابتة app.use(express.json()); app.use(express.urlencoded({ extended: true })); app.use(express.static(path.join(__dirname, 'public'))); app.use(express.static(__dirname)); // التوجيهات الرئيسية للتوثيق والصفحات (Routes) app.get('/', (req, res) => { res.sendFile(path.join(__dirname, 'login.html'), (err) => { if (err) { res.sendFile(path.join(__dirname, 'public', 'login.html')); } }); }); app.get('/login', (req, res) => { res.sendFile(path.join(__dirname, 'login.html'), (err) => { if (err) { res.sendFile(path.join(__dirname, 'public', 'login.html')); } }); }); app.get('/login.html', (req, res) => { res.sendFile(path.join(__dirname, 'login.html'), (err) => { if (err) { res.sendFile(path.join(__dirname, 'public', 'login.html')); } }); }); app.get('/admin', (req, res) => { res.sendFile(path.join(__dirname, 'admin.html'), (err) => { if (err) { res.sendFile(path.join(__dirname, 'public', 'admin.html')); } }); }); app.get('/admin-login', (req, res) => { res.sendFile(path.join(__dirname, 'admin-login.html'), (err) => { if (err) { res.sendFile(path.join(__dirname, 'public', 'admin-login.html')); } }); }); // تشغيل الخادم app.listen(PORT, () => { console.log(`Server is running on port ${PORT}`); });
